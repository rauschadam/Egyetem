// 1) Eseményargumentum: külön fájl a modellprojektben
public class FieldChangedEventArgs : EventArgs
{
    public int X { get; }
    public int Y { get; }

    public FieldChangedEventArgs(int x, int y)
    {
        X = x;
        Y = y;
    }
}

// 2) Deklaráció a modellosztályban
public event EventHandler<FieldChangedEventArgs>? FieldChanged;
public event EventHandler? TableChanged;                   // ha nem visz adatot
public event EventHandler<GameOverEventArgs>? GameOver;

// 3) Kiváltás: a ?. miatt nem száll el, ha senki sem iratkozott fel
private void OnFieldChanged(int x, int y)
    => FieldChanged?.Invoke(this, new FieldChangedEventArgs(x, y));

private void OnTableChanged()
    => TableChanged?.Invoke(this, EventArgs.Empty);

// 4) A nézet csak olvashat a táblából: indexelő
public int this[int x, int y] => _table[x, y];

Nézet (GameForm.cs): feliratkozás + kezelők

private readonly GameModel _model;

public GameForm()
{
    InitializeComponent();

    _model = new GameModel();
    _model.FieldChanged += Model_FieldChanged;      // ① feliratkozás
    _model.TableChanged += Model_TableChanged;
    _model.TimeAdvanced += Model_TimeAdvanced;
    _model.GameOver += Model_GameOver;

    _timer.Tick += Timer_Tick;

    GenerateTable();                                // ② gombok
    StartNewGame(Difficulty.Easy);                  // ③ CSAK EZUTÁN új játék!
}

// Kezelő alakja: private void Név(object? sender, XxxEventArgs e)
private void Model_FieldChanged(object? sender, FieldChangedEventArgs e)
{
    _buttons[e.X, e.Y].BackColor = ToColor(_model[e.X, e.Y]);
}

2Gombrács létrehozása kódból

A Designerben csak egy üres TableLayoutPanel kell (_tablePanel, Dock = Fill). A sorokat, oszlopokat és gombokat kód hozza létre:

private readonly Button[,] _buttons = new Button[GameModel.TableSize, GameModel.TableSize];

private void GenerateTable()
{
    int n = GameModel.TableSize;
    _tablePanel.SuspendLayout();                 // ne rajzolja újra 225-ször
    _tablePanel.Controls.Clear();
    _tablePanel.ColumnStyles.Clear();            // a Designerben beállított vegyes stílusok törlése
    _tablePanel.RowStyles.Clear();
    _tablePanel.ColumnCount = n;
    _tablePanel.RowCount = n;
    for (int k = 0; k < n; k++)                  // MIND egyforma: 100/n százalék
    {
        _tablePanel.ColumnStyles.Add(new ColumnStyle(SizeType.Percent, 100F / n));
        _tablePanel.RowStyles.Add(new RowStyle(SizeType.Percent, 100F / n));
    }

    for (int x = 0; x < n; x++)                  // x = sor
    {
        for (int y = 0; y < n; y++)              // y = oszlop
        {
            Button button = new()
            {
                Dock = DockStyle.Fill,               // kitölti a cellát
                Margin = new Padding(0),             // nincs rés a gombok között
                FlatStyle = FlatStyle.Flat,          // lapos, egyszínű kitöltés
                BackColor = Color.White,             // a színt később a modell eseménye állítja
                Text = string.Empty,                 // ha számot/betűt kell kiírni
                Font = new Font(Font, FontStyle.Bold),
                Tag = (x, y),                        // KELL: ebből tudjuk, melyikre kattintottak
                TabStop = false,
            };
            button.FlatAppearance.BorderColor = Color.Gray;
            button.Click += FieldButton_Click;       // EGY közös kezelő mind a 225 gombnak

            _buttons[x, y] = button;
            _tablePanel.Controls.Add(button, y, x);  // (vezérlő, OSZLOP, SOR), nem (sor, oszlop)!
        }
    }

    _tablePanel.ResumeLayout();
}

// Kattintás: a koordináta a Tagből jön, a szabályt a MODELL dönti el
private void FieldButton_Click(object? sender, EventArgs e)
{
    if (sender is Button { Tag: (int x, int y) })
    {
        _model.Step(x, y);
    }
}

3Időzítő (Timer)

A nézetben egy WinForms Timer ketyeg, minden Tick a modell AdvanceTime()-ját hívja. A modell számolja az időt, és eseménnyel szól a nézetnek. Így nincs szálkezelési gond.

Nézet

// mezőként; a TELJES név kell (különben CS0104: Timer is ambiguous)
private readonly System.Windows.Forms.Timer _timer = new() { Interval = 1000 };   // ezredmásodperc

// konstruktorban:
_timer.Tick += Timer_Tick;

private void Timer_Tick(object? sender, EventArgs e) => _model.AdvanceTime();

private void Model_TimeAdvanced(object? sender, EventArgs e)
{
    _timeLabel.Text = $"Idő: {TimeSpan.FromSeconds(_model.GameTime):mm\\:ss}";   // Idő: 01:05
}

_timer.Start();                      // új játéknál
_timer.Stop();                       // játék vége
_timer.Enabled = !_timer.Enabled;    // szünet ki/be

// szünet menüpont/gomb
private void MenuPause_Click(object? sender, EventArgs e)
{
    _timer.Enabled = !_timer.Enabled;
    SetTableEnabled(_timer.Enabled);                          // szünetben ne lehessen lépni
    _statusLabel.Text = _timer.Enabled ? "Fut" : "Szünet";
}

Modell

private int _gameTime;
public int GameTime => _gameTime;
public event EventHandler? TimeAdvanced;

public void AdvanceTime()
{
    if (IsGameOver)
    {
        return;
    }

    _gameTime++;
    TimeAdvanced?.Invoke(this, EventArgs.Empty);
}
// NewGame-ben: _gameTime = 0;

private void Model_GameOver(object? sender, GameOverEventArgs e)
{
    _timer.Stop();                // ① az idő megáll (a MessageBox alatt is ketyegne!)
    SetTableEnabled(false);       // ② több lépés nincs

    DialogResult result = MessageBox.Show(                                   // ③ kérdés
        $"Vége a játéknak!{Environment.NewLine}Pontszám: {e.Score}{Environment.NewLine}Új játék?",
        "SameGame",
        MessageBoxButtons.YesNo,
        MessageBoxIcon.Question);

    if (result == DialogResult.Yes)
    {
        StartNewGame(_model.Difficulty);
    }
    else
    {
        Close();
    }
}

// Kilépés megerősítése: konstruktorban (vagy ⚡ Events → FormClosing)
FormClosing += GameForm_FormClosing;

private void GameForm_FormClosing(object? sender, FormClosingEventArgs e)
{
    if (MessageBox.Show("Biztos kilépsz?", "Kilépés", MessageBoxButtons.YesNo, MessageBoxIcon.Question)
        == DialogResult.No)
    {
        e.Cancel = true;          // nem zárul be
    }
}

5Új játék + nehézség menüből

private void StartNewGame(Difficulty difficulty)
{
    SetTableEnabled(true);
    _statusLabel.Text = "Új játék";
    _timeLabel.Text = "Idő: 00:00";
    _model.NewGame(difficulty);      // TableChanged → a gombok átszíneződnek
    _timer.Start();
}

private void MenuEasy_Click(object? sender, EventArgs e) => StartNewGame(Difficulty.Easy);
private void MenuMedium_Click(object? sender, EventArgs e) => StartNewGame(Difficulty.Medium);
private void MenuHard_Click(object? sender, EventArgs e) => StartNewGame(Difficulty.Hard);
private void MenuExit_Click(object? sender, EventArgs e) => Close();

public enum Difficulty { Easy = 3, Medium = 5, Hard = 7 }   // modell; (int)difficulty = színek száma
